  * _id : ObjectId <<PK>>
  --
  ruleName : String
  ruleContent : String
  version : Integer
  ruleGroup : String
  errorCodeId : ObjectId <<FK>>
  active : Boolean
  createdAt : Date
  updatedAt : Date
}

entity "rules_staging" {
  * _id : ObjectId <<PK>>
  --
  ruleName : String
  ruleContent : String
  version : Integer
  ruleGroup : String
  errorCodeId : ObjectId <<FK>>
  staged : Boolean
  createdAt : Date
  updatedAt : Date
}

  * _id : ObjectId <<PK>>
  --
  flowType : String
  ruleIds : Array[ObjectId] <<FK>>
  priority : Integer
  active : Boolean
  updatedAt : Date
}

entity "client_rule_mappings" {
  * _id : ObjectId <<PK>>
  --
  clientId : String
  ruleId : ObjectId <<FK>>
  enabled : Boolean
  priority : Integer
  updatedAt : Date
}

  * _id : ObjectId <<PK>>
  --
  clientId : String
  allRulesEnabled : Boolean
  updatedAt : Date
}

  * _id : ObjectId <<PK>>
  --
  errorCode : Integer
  errorMessage : String
  errorCategory : String
  active : Boolean
  createdAt : Date
  updatedAt : Date
}
